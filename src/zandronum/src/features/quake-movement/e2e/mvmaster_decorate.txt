// [rc4l] Master E2E pawn matrix for features/quake-movement. See MASTERTEST.md.
//
// One class per behaviour, deliberately: a pawn carrying several movement flags at once cannot
// tell you WHICH flag produced an observed jump. Every class here is reachable by setting the
// client's `playerclass` CVAR and restarting the map, and `dumpactor` reports the class back so a
// sample can never be silently attributed to the wrong pawn.

// A visible, harmless puff for the effect-actor slots. CLIENTSIDEONLY because that is how the
// feature spawns them -- if this ever gets replicated, this actor is the thing that would prove it.
ACTOR MvDust
{
	Radius 2
	Height 2
	+NOBLOCKMAP
	+NOGRAVITY
	+CLIENTSIDEONLY
	RenderStyle Translucent
	Alpha 0.6
	States
	{
	Spawn:
		PUFF ABCD 3
		Stop
	}
}

// ---------------------------------------------------------------------------------------------
// Jump matrix. JumpDelay 20 on every one of these is load-bearing: a second jump leaves jumpTics
// positive, which is what distinguishes it from a main jump in a dumpactor sample.
// ---------------------------------------------------------------------------------------------
ACTOR MvJump : MvQBase
{
	Player.SecondJumpAmount 1
	Player.SecondJumpZ 10
	Player.SecondJumpDelay 0
	Player.JumpDelay 20
	Player.DisplayName "MvJump"
}

// A JumpDelay far from every other timing constant in the engine, so the ground phase between two
// held-jump bounces names its own source: 45 tics can be confused with nothing else.
ACTOR MvJumpSlow : MvJump
{
	Player.JumpDelay 45
	Player.DisplayName "MvJumpSlow"
}

// SecondJumpAmount is tri-state; -1 is unlimited, and that is a separate code path from a count.
ACTOR MvJumpInf : MvJump
{
	Player.SecondJumpAmount -1
	Player.DisplayName "MvJumpInf"
}

// Unlimited second jumps but a long cooldown between them, so SecondJumpDelay is measurable as the
// gap between consecutive second jumps rather than being masked by the one-shot allowance.
ACTOR MvSJDelay : MvJumpInf
{
	Player.SecondJumpDelay 25
	Player.DisplayName "MvSJDelay"
}

// +ABSOLUTESECONDJUMP SETS horizontal velocity instead of adding to it. SecondJumpXY 6 is well
// under the 13.141 terminal speed, so "set" and "add" land on unmistakably different numbers:
// 6.0 with the flag, ~19.1 without.
ACTOR MvAbsSJ : MvJump
{
	+ABSOLUTESECONDJUMP
	Player.SecondJumpXY 6
	Player.DisplayName "MvAbsSJ"
}

// The paired control: identical apart from the flag, so the comparison is one bit wide.
ACTOR MvAbsSJOff : MvJump
{
	Player.SecondJumpXY 6
	Player.DisplayName "MvAbsSJOff"
}

ACTOR MvWallJump : MvJump

	+WALLJUMP
	Player.DisplayName "MvWallJump"

// V2 pushes along the WALL NORMAL rather than the way the player faces. Running east along a wall

// ---------------------------------------------------------------------------------------------
// Runtime mutation, for the two wire paths this feature adds. Given with the `give` cheat, which
// a client routes to the server (CLC_GIVECHEAT), so A_ChangeFlag runs SERVER-side on the pawn and
// the resulting SERVERCOMMANDS_SetThingFlags(FLAGSET_MVFLAGS) is a real replication.
// ---------------------------------------------------------------------------------------------
ACTOR MvGiveSlideOn : CustomInventory
{
	-COUNTITEM
	Inventory.MaxAmount 0
	States
	{
	Spawn:
		TNT1 A -1
		Stop
	Pickup:
		TNT1 A 0 A_ChangeFlag("CROUCHSLIDE", 1)
		Stop
	}
}

ACTOR MvGiveSlideOff : MvGiveSlideOn
{
	States
	{
	Pickup:
		TNT1 A 0 A_ChangeFlag("CROUCHSLIDE", 0)
		Stop
	}
}
