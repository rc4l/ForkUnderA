# fulldamagedistance are plain map units, as P_RadiusAttack takes them.

	ExtendedCommand
	AproxFixed x
	AproxFixed y
	AproxFixed z
	Short distance
	Short fulldamagedistance
EndCommand

# [rc4l] Sprite orientation -- roll now, SpriteAngle/SpriteRotation later through the same flag
# byte. Defined HERE, in the last spec file the generator reads, because command ids are assigned in
# spec-file order rather than from network_enums.h: putting this in spec.things.txt inserted it at
# id 26 and shifted every command after it, which protocol-snapshot caught.
Command SetThingSpriteOrientation
	ExtendedCommand
	Actor actor
	Byte orientflags

	If (orientflags & SPRITEORIENT_ROLL)
		AproxAngle roll
	EndIf
EndCommand

# [Doruk-Aksoy] ACS fua_SetShaderUniform1f..4f and 1i: one uniform of a GLDEFS post-process shader, kind 0 one int, 1 to 4 that many floats.
Command FuaSetShaderUniform
	ExtendedCommand
	String shader
	String uniform
	Bool own
	Byte kind
	If (kind == 0)
		Long intValue
	EndIf
	If (kind >= 1)
		Float x
	EndIf
	If (kind >= 2)
		Float y
	EndIf
	If (kind >= 3)
		Float z
	EndIf
	If (kind >= 4)
		Float w
	EndIf
EndCommand
